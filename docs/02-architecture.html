1. مبادئ التصميم

  1. مسار البلاغ أولا: استقبال البلاغ وإسناد رقم التتبع لا يتوقفان على أي خدمة ذكاء اصطناعي أو بث. كل معالجة ذكية تتم بشكل غير متزامن بعد الحفظ.
  2. وحدات مستقلة داخل نظام واحد قابل للتفكيك: نبدأ بخلفية واحدة منظمة في وحدات ذات حدود واضحة (Modular Monolith) مع خدمات منفصلة فقط حيث يفرض ذلك نوع العمل (الذكاء الاصطناعي، البث، الهاتف). يمكن فصل أي وحدة لاحقا دون إعادة كتابة.
  3. مفتوح المصدر ومحلي: كل المكونات مجانية وقابلة للتشغيل داخل مركز بيانات تونسي، ولا تخرج المعطيات الشخصية نحو خدمات أجنبية.
  4. الإنسان يقرر: النماذج تقترح وتشرح، والمشغّل أو القائد يصادق.
  5. الخصوصية منذ التصميم: تقليل المعطيات، تشفير الحقول الحساسة، فصل هوية المبلّغ المجهول.
  6. قابلية التوسع التدريجي: نفس الكود يعمل على خادم واحد في التجربة وعلى عنقود في النطاق الوطني.

5.1 المخططات (Schemas)

قاعدة PostgreSQL واحدة مقسمة إلى مخططات حسب الوحدة، بحيث لا تكتب وحدة في جداول وحدة أخرى مباشرة:

identity · reports · ops · orgs · geo · notify · content · rewards · knowledge · audit

5.2 مقتطف من الجداول الأساسية

CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS vector;
CREATE EXTENSION IF NOT EXISTS pgcrypto;

CREATE TABLE geo.gouvernorat (
  id smallint PRIMARY KEY,               -- 1..24
  code text UNIQUE NOT NULL,
  name_ar text NOT NULL, name_fr text NOT NULL,
  boundary geometry(MultiPolygon, 4326)
);
CREATE TABLE geo.delegation (
  id integer PRIMARY KEY,
  gouvernorat_id smallint NOT NULL REFERENCES geo.gouvernorat(id),
  name_ar text NOT NULL, name_fr text NOT NULL,
  boundary geometry(MultiPolygon, 4326)
);

CREATE TYPE identity.account_kind AS ENUM ('citizen','organization','anonymous');
CREATE TABLE identity.account (
  user_id uuid PRIMARY KEY DEFAULT gen_random_uuid(),   -- USERID الموحد
  kind identity.account_kind NOT NULL,
  keycloak_sub text UNIQUE,                             -- فارغ للمجهول
  anon_pseudonym bytea UNIQUE,                          -- HMAC لمعرف الجهاز، للمجهول فقط
  status text NOT NULL DEFAULT 'active',                -- active | suspended | banned
  false_report_count int NOT NULL DEFAULT 0,
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE identity.profile (
  user_id uuid PRIMARY KEY REFERENCES identity.account(user_id),
  first_name text, last_name text,
  cin_enc bytea,                 -- رقم بطاقة التعريف مشفّر على مستوى الحقل
  cin_hash bytea UNIQUE,         -- للبحث دون فك التشفير
  phone text, email text,
  gouvernorat_id smallint REFERENCES geo.gouvernorat(id),
  delegation_id int REFERENCES geo.delegation(id),
  address text,
  medical_enc bytea,             -- العمر، الفصيلة، المعطيات الطبية (JSON مشفّر)
  locale text NOT NULL DEFAULT 'ar'
);

CREATE TABLE reports.report (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tracking_code text UNIQUE NOT NULL,    -- مثال: EH-26-SFX-0004821
  client_uuid uuid UNIQUE,               -- لمنع التكرار عند إعادة الإرسال دون اتصال
  user_id uuid NOT NULL REFERENCES identity.account(user_id),
  incident_id uuid,
  category text NOT NULL,
  text_original text, lang_original text, text_ar text,
  location geography(Point, 4326) NOT NULL,
  location_accuracy_m real,
  gouvernorat_id smallint REFERENCES geo.gouvernorat(id),
  priority_score smallint, credibility_score smallint,
  status text NOT NULL DEFAULT 'received',
  source text NOT NULL DEFAULT 'app',    -- app | portal | iot | call | sms
  embedding vector(768),
  created_at timestamptz NOT NULL, received_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ON reports.report USING gist (location);
CREATE INDEX ON reports.report (gouvernorat_id, status, created_at DESC);

CREATE TABLE reports.media (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  report_id uuid REFERENCES reports.report(id),
  stream_id uuid,
  kind text NOT NULL,                    -- image | video | audio | recording
  object_key text NOT NULL,              -- مسار في تخزين S3
  sha256 char(64) NOT NULL,
  bytes bigint, duration_s real,
  transcript text, analysis jsonb,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE ops.incident (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  category text NOT NULL, severity smallint,
  centroid geography(Point, 4326), area geography(Polygon, 4326),
  gouvernorat_id smallint REFERENCES geo.gouvernorat(id),
  op_level text NOT NULL DEFAULT 'normal',   -- normal | crisis | disaster
  status text NOT NULL DEFAULT 'open',
  opened_at timestamptz NOT NULL DEFAULT now(), closed_at timestamptz
);
CREATE TABLE ops.timeline_event (
  id bigserial PRIMARY KEY,
  incident_id uuid NOT NULL REFERENCES ops.incident(id),
  at timestamptz NOT NULL DEFAULT now(),
  actor uuid, kind text NOT NULL, payload jsonb
);

CREATE TABLE audit.log (
  id bigserial PRIMARY KEY,
  at timestamptz NOT NULL DEFAULT now(),
  actor uuid, action text NOT NULL, target text, reason text,
  prev_hash char(64), hash char(64) NOT NULL     -- سلسلة بصمات تمنع التعديل
);

6. طبقة الذكاء الاصطناعي

كل النماذج تعمل محليا. خادم GPU واحد متوسط (مثلا 24 GB VRAM) يكفي للتجربة، ويمكن تشغيل الكاشف البصري على أجهزة طرفية.

القدرةالنموذج المقترحبديلالمدخل والمخرج
تحويل الصوت إلى نصWhisper large-v3 عبر faster-whisperWhisper medium على المعالجتسجيل حتى 5 دقائق ← نص ولغة. تحسين لاحق باللهجة التونسية عبر ضبط دقيق على بيانات مُعلّمة.
الترجمةOpus-MT fr→ar و en→arالنموذج اللغوي المحلينص ← نص عربي مع الاحتفاظ بالأصل.
التصنيف والاستخلاصنموذج لغوي مفتوح 7 إلى 8 مليار معامل عبر Ollama (Qwen2.5 أو Llama 3.1)مصنف خفيف مدرّب على البلاغاتنص ← نوع الحادثة، مؤشرات وجود أرواح في خطر، عدد المصابين، كلمات مفتاحية، بصيغة JSON مضبوطة.
التجميع الدلالينموذج تضمين متعدد اللغات (multilingual-e5 أو bge-m3)—نص ← متجه في pgvector، يُستعمل مع المسافة والزمن.
الأولويةقواعد شفافة موزونة (BR-20) ثم نموذج إحصائي بعد تجميع بيانات كافية—ميزات البلاغ ← درجة 0 إلى 100 مع تفسير مكوناتها.
كشف الحرائق والدخانYOLO مدرّب على بيانات حرائق ودخانRT-DETRصورة أو إطارات البث ← صناديق وثقة.
انتشار الحريقنموذج فيزيائي مبسط (Rothermel) مع الرياح والرطوبة والانحدارتكامل لاحق مع أدوات مثل FARSITEنقطة الاشتعال والطقس ← مضلعات انتشار متوقعة.
المساعد لدعم القرارالنموذج اللغوي المحلي مع استرجاع من الذاكرة الوطنية ودليل الإجراءات (RAG)—حالة الحادثة ← توصيات مع مصادرها.
رخص النماذج: مكتبة Ultralytics (YOLOv8 و YOLO11) مرخصة AGPL-3.0، وتشترط نشر الكود المصدري أو اقتناء رخصة تجارية. البديل المقترح هو تنفيذ MultimediaTechLab/YOLO المرخص MIT، أو RT-DETR المرخص Apache-2.0. كذلك نموذج الترجمة NLLB-200 مرخص للاستعمال غير التجاري فقط ولذلك لم يُعتمد. يُراجع ترخيص كل نموذج لغوي قبل الاعتماد.

7. مسار البث والرؤية الحاسوبية

تطبيق الهاتف ── WHIP/WebRTC ──┐
درون / كاميرا ── RTSP/RTMP ────┼──► MediaMTX ── runOnReady / runOnNotReady (Webhook) ──► خلفية ء-حماية
                               │        │                                              (ربط البث بالمستخدم والبلاغ)
                               │        ├── record: fMP4 ──► S3 ──► SHA-256 ──► reports.media
                               │        ├── RTSP داخلي ──► كاشف الحرائق والدخان ──► media.analyzed
                               │        └── WebRTC / HLS ──► لوحة التحكم (عرض متعدد الشاشات)
  • المصادقة على البث عبر authHTTPAddress في MediaMTX، الذي يسأل الخلفية إن كان للمستخدم بلاغ مفتوح ورمز صالح.
  • التسجيل الآلي بقطع زمنية، ثم حساب البصمة عند إغلاق كل قطعة وربطها بالبلاغ (FR-C-07).
  • الكاشف يقرأ إطارا أو اثنين في الثانية لكل بث لتخفيف الحمل، ويرفع المعدل عند الاشتباه.
  • مشاريع جاهزة للتقييم في المرحلة 6: SpectraX (يجمع MediaMTX مع YOLO) ومشاريع كشف الحرائق والدخان عبر RTSP المنشورة على GitHub و Hugging Face. يُقيّم كل منها على الدقة والرخصة وقابلية الصيانة قبل الاعتماد، ويمكن أن نأخذ منها الأفكار ونبني كاشفا خاصا برخصة مناسبة.
  • التشغيل الطرفي: تصدير الكاشف بصيغة ONNX أو NCNN للعمل على Raspberry Pi 5 قرب الكاميرات، وإرسال النتائج فقط.

9. العمل دون اتصال والمزامنة (تطبيق الهاتف)

  • قاعدة محلية SQLite مشفرة (SQLCipher عبر drift) تحفظ البلاغات والوسائط ونصائح السلامة وأرقام الطوارئ.
  • كل بلاغ يحمل client_uuid يُنشأ على الجهاز، فيكون الإرسال المتكرر غير مضاعف (idempotent) على الخادم.
  • طابور إرسال يعمل في الخلفية (WorkManager على أندرويد و BGTaskScheduler على iOS) ويعيد المحاولة بفواصل متزايدة.
  • رفع الوسائط بالاستئناف (بروتوكول tus) لأن الشبكات الضعيفة شائعة أثناء الكوارث، مع ضغط الصور قبل الرفع والاحتفاظ بالأصل.
  • رقم تتبع مؤقت محلي يُستبدل بالرقم النهائي عند التأكيد، ويُعرض للمستخدم في الحالتين.
  • عند غياب الإنترنت تماما: اقتراح الاتصال بـ198 أو إرسال SMS مُنسّق يحتوي النوع والإحداثيات.

10. هيكلة الأمن

الطبقةالإجراءات
الهويةKeycloak: OIDC مع PKCE للتطبيق والواب، MFA إجباري لكل مستخدمي اللوحة، سياسات كلمات المرور، قفل الحساب، CAPTCHA بعد 3 محاولات، OTP بالهاتف أو البريد.
الصلاحياتRBAC من Keycloak مع نطاق جغرافي (ولاية، معتمدية) ومستوى تشغيلي، تُفرض في الخلفية على كل استعلام (وليس في الواجهة فقط)، مع أمن على مستوى الصفوف في PostgreSQL للبيانات الحساسة.
النقلTLS 1.3 خارجيا، و mTLS بين الخدمات في النطاق الوطني، وتثبيت الشهادة في تطبيق الهاتف.
التخزينتشفير الأقراص، تشفير التخزين من جهة الخادم، تشفير الحقول الحساسة بمفاتيح منفصلة تُدار في OpenBao (بديل حر لـ Vault).
المجهولالمعرف المستعار = HMAC(سر الخادم، معرف التثبيت)، لا يُخزن IP (البوابة تحذفه من السجلات لمسار المجهول)، إزالة بيانات EXIF والموقع المضمّن من الوسائط بعد استخراج ما يلزم.
الحماية من الإساءةتحديد معدل الطلبات حسب الحساب والجهاز، حماية ضد التعداد (ردود موحدة)، فحص الملفات المرفوعة (النوع، الحجم، مضاد الفيروسات ClamAV).
التدقيقسجل إلحاق فقط بسلسلة بصمات، تسجيل كل اطلاع على المعطيات الطبية.
التطوير الآمنفحص التبعيات والأسرار في التكامل المستمر، تحليل ساكن (Semgrep، Bandit)، اختبار OWASP ZAP، اختبار اختراق قبل الإطلاق.
